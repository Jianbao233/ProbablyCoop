using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;
using ProbablyCoop.Api;
using UnityEngine;

namespace ProbablyCoopExtensionProbe;

public sealed class ProbeMod : IMod
{
    private const string ModId = "probably_coop_extension_probe";
    private const string StorageProbeName = "Co-op declared storage probe";
    private const string CareerId = "cooperative_brewer_fixture";
    private const string CareerItem = "Co-op brewer fixture";
    private const int CareerCost = 430;
    private const string PersonalPerkId = "coop_probe_personal_perk";
    private const string SharedPerkId = "coop_probe_shared_perk";
    private bool careerRegistered;
    private CoopExtensionSession session;
    private GameObject runner;
    private int count;
    private int displayed = -1;
    private int phase;
    private int localActivations;
    private bool sawHostSession;
    private ulong firstRevision;
    private bool forceReconnect;
    private bool expeditionProbe, projectedAtExchange;
    private ulong exchangeRevision;

    public void Init(ModManifest manifest) { }

    public void OnEnable()
    {
        ModHelper.OnMessage(ModId,CoopExtensionApi.DiscoveryChannel,Discover);
        ModHelper.Expose(ModId, CoopExtensionApi.LocalItemActivationService, (Func<GameItem, bool>)OpenStorageView);
        var arguments = Environment.GetCommandLineArgs();
        if (!Application.productName.StartsWith("Probably Stolen Coop RegressionRev12Water ", StringComparison.Ordinal))
            return;
        if (arguments.Contains("--probably-coop-qa-mod-opening"))
        {
            ModHelper.AddStartingScenario(ModId, CareerId, () => "QA brewer",
                () => "Supplies and a cooperative utility grant.",
                new Func<string>[] { () => "Juice and a utility unlock" }, ApplyOpeningCareer, skipIntro: true);
            ModHelper.Expose(ModId, CoopExtensionApi.OpeningCareersService,
                new[] { new CoopOpeningCareer(CareerId, CareerCost) });
            ModHelper.AddStartingPerk(ModId, PersonalPerkId, 0, StartingPerk.StartingPerkType.NEUTRAL,
                () => "QA personal trait", () => "Only active for the player that picked it.");
            ModHelper.AddStartingPerk(ModId, SharedPerkId, 0, StartingPerk.StartingPerkType.NEUTRAL,
                () => "QA shop trait", () => "Active for both players in this shop.");
            ModHelper.Expose(ModId, CoopExtensionApi.OpeningPerksService, new Dictionary<string, CoopOpeningPerkScope>
            {
                [PersonalPerkId] = CoopOpeningPerkScope.Personal,
                [SharedPerkId] = CoopOpeningPerkScope.SharedShop
            });
            ModHook.OnGameLoadedEarly += VerifyLoadedOpening;
            careerRegistered = true;
        }
        expeditionProbe = arguments.Contains("--probably-coop-qa-extension-expedition");
        if (!expeditionProbe && !arguments.Contains("--probably-coop-qa-extension-probe")) return;
        forceReconnect = arguments.Contains("--probably-coop-qa-extension-rejoin");
        runner = new GameObject("ProbablyCoopExtensionProbe");
        UnityEngine.Object.DontDestroyOnLoad(runner);
        runner.AddComponent<ProbePump>().TickAction = Update;
    }

    public void OnDisable()
    {
        ModHelper.OffMessage(ModId,CoopExtensionApi.DiscoveryChannel,Discover);
        ModHelper.Expose(ModId, CoopExtensionApi.LocalItemActivationService, null);
        if (careerRegistered)
        {
            ModHook.OnGameLoadedEarly -= VerifyLoadedOpening;
            ModHelper.Expose(ModId, CoopExtensionApi.OpeningCareersService, null);
            ModHelper.RemoveStartingScenarios(ModId);
            ModHelper.Expose(ModId, CoopExtensionApi.OpeningPerksService, null);
            ModHelper.RemoveStartingPerks(ModId);
            careerRegistered = false;
        }
        if(runner!=null) UnityEngine.Object.Destroy(runner);
        runner=null;
        session=null;
    }

    private static void ApplyOpeningCareer()
    {
        if (ModHelper.GetStartingScenarioId() != CareerId || !ModHelper.IsStartingScenario(CareerId))
            throw new InvalidOperationException("The native callback has the wrong current career.");
        if (!ModHelper.IsStartingScenario("water_merchant") || !PlayerStore.Instance.IsWaterMerchantStart())
            throw new InvalidOperationException("The native callback lost the other player's water merchant career.");
        Debug.Log("[coop-extension-probe] PASS opening callback sees both careers");
        var item = DirectoryMaster.Item("processed_juice");
        GeneralHelper.SetCustomName(item, CareerItem);
        GraphUtils.EmporiumTryAdd(EmporiumEntry.Instance.invElement, item);
        PlayerStore.Instance.playerCash -= CareerCost;
        NetworkUpgrade.GetUpgradeById("GUTTERFLOW_TAP").Unlock();
    }

    private static void VerifyLoadedOpening()
    {
        string primary = ModHelper.GetStartingScenarioId();
        if (primary != "water_merchant" && primary != CareerId) return;
        if (!ModHelper.IsStartingScenario(CareerId) || !ModHelper.IsStartingScenario("water_merchant") ||
            !PlayerStore.Instance.IsWaterMerchantStart() || !ModHelper.IsStartingPerkActive(SharedPerkId) ||
            ModHelper.IsStartingPerkActive(PersonalPerkId))
            throw new InvalidOperationException("The early load callback lost the saved partner career or shared perk.");
        Debug.Log("[coop-extension-probe] PASS early load sees both careers and the shared perk");
    }

    private void Discover(string sender,object payload)
    {
        if(sender!="probably_coop") return;
        var registration = new CoopExtensionRegistration("counter-v1",new[] { "increment" },
            ReadState,Execute,Project,Attach);
        ModHelper.SendMessage(ModId,CoopExtensionApi.RegistrationChannel,registration);
    }

    private string ReadState()
    {
        if(session?.IsReady==true)
            throw new InvalidOperationException("An extension session must stay unavailable inside ReadState.");
        if (expeditionProbe && MapUIManager.Instance?.Current?.Id == AfterhourLocation.COMMISSARY)
            throw new InvalidOperationException("Shared state read followed the guest into the exchange.");
        return count.ToString(CultureInfo.InvariantCulture);
    }

    private void Attach(CoopExtensionSession next)
    {
        session=next;
        phase=0;
        localActivations=0;
        displayed=-1;
        firstRevision=0;
        projectedAtExchange=false;
        exchangeRevision=0;
        sawHostSession=false;
        Debug.Log("[coop-extension-probe] attached role="+next.LocalActor);
    }

    private CoopActionResult Execute(CoopExtensionAction action)
    {
        if(action.Id!="increment" || action.Payload!="1" || count>=1000) return CoopActionResult.Rejected;
        if (expeditionProbe && (action.Actor != CoopExtensionActor.Partner ||
            MapUIManager.Instance?.Current?.Id != AfterhourLocation.COMMISSARY))
            throw new InvalidOperationException("The expedition action lost its native guest location.");
        count++;
        Debug.Log("[coop-extension-probe] authoritative increment actor="+action.Actor+" count="+count);
        return CoopActionResult.Applied;
    }

    private void Project(CoopExtensionState state)
    {
        displayed=int.Parse(state.Payload,CultureInfo.InvariantCulture);
        Debug.Log("[coop-extension-probe] guest state count="+displayed+" revision="+state.Revision);
        if (!expeditionProbe) return;
        string site = MapUIManager.Instance.Current?.Id;
        if (site == AfterhourLocation.COMMISSARY)
        {
            if (!projectedAtExchange && (state.Revision != firstRevision || displayed != 0))
                throw new InvalidOperationException("The exchange view did not get the unchanged entry state.");
            if (!projectedAtExchange)
                Debug.Log("[coop-extension-probe] PASS unchanged state reached the new guest scene");
            projectedAtExchange = true;
            exchangeRevision = state.Revision;
        }
        else if (site == null && projectedAtExchange)
        {
            if (state.Revision != exchangeRevision || displayed != 1)
                throw new InvalidOperationException("The shop did not recover the confirmed expedition state.");
            Debug.Log("[coop-extension-probe] PASS unchanged state rebound after guest return");
            projectedAtExchange = false;
        }
        else if (site == null && firstRevision == 0)
            firstRevision = state.Revision;
    }

    private void Update()
    {
        if(session?.IsClosed==true) session=null;
        if(session?.IsReady!=true) return;
        if (expeditionProbe)
        {
            TickExpeditionProbe();
            return;
        }
        if(session.LocalActor==CoopExtensionActor.Host)
        {
            if(!sawHostSession)
            {
                if (!InstallStorageFixture()) return;
                Debug.Log("[coop-extension-probe] PASS native host discovery");
                sawHostSession=true;
            }
            return;
        }
        var state=session.CurrentState;
        if(state==null || phase!=0) return;
        if (!VerifyStorageFixture()) return;
        Debug.Log("[coop-extension-probe] PASS custom wrapper declared native slot order and ownership");
        if(forceReconnect)
        {
            if(state.Payload=="1" && displayed==1)
            {
                Debug.Log("[coop-extension-probe] PASS native reconnect recovered count=1");
                phase=3;
            }
            return;
        }
        if(state.Payload!="0" || displayed!=0) return;
        firstRevision=state.Revision;
        if(session.Request("increment","1",firstRevision,ConfirmIncrement)!=CoopActionSubmission.Queued) return;
        phase=1;
    }

    private void TickExpeditionProbe()
    {
        if (session.LocalActor != CoopExtensionActor.Partner || !projectedAtExchange || phase != 0) return;
        var state = session.CurrentState;
        if (state == null || state.Payload != "0") return;
        var submission = session.Request("increment", "1", state.Revision, result =>
        {
            if (result == CoopActionResult.Busy) { phase = 0; return; }
            if (result != CoopActionResult.Applied || displayed != 1)
                Debug.LogError("[coop-extension-probe] FAIL expedition action returned " + result);
            else Debug.Log("[coop-extension-probe] PASS guest-location action and projected receipt");
            phase = 2;
        });
        if (submission == CoopActionSubmission.Queued) phase = 1;
    }

    private static bool InstallStorageFixture()
    {
        var shop = EmporiumEntry.Instance;
        if (shop?.invElement == null) return false;
        var item = DirectoryMaster.Item("labeler");
        ModHelper.SetSource(item, ModId, "ProbablyCoop Extension Probe");
        item.mayActivateSlotItemFunc = (_, _, _) => true;
        item.canActivateSlotItemFunc = (_, _, _) => true;
        item.onActivateSlotItemFunc = (_, _, _) =>
            Debug.LogError("[coop-extension-probe] FAIL local opening was sent to host");
        var layout = new DeclaredStorage();
        var window = new PixelWindow(true, StorageProbeName);
        window.Attach(layout);
        item.SetContentWindow(window);
        GeneralHelper.SetItemOwned(item, true);
        GeneralHelper.SetCustomName(item, StorageProbeName);
        var water = DirectoryMaster.Item("small_bottled_water");
        GeneralHelper.SetItemOwned(water, true);
        if (GraphUtils.TryAcceptAll(layout.CoopStorage[1], water) != 0 ||
            (GraphUtils.TryAcceptAll(shop.invElement, item) != 0 && GraphUtils.TryAcceptAll(shop.backInvinvElement, item) != 0))
            throw new InvalidOperationException("Could not place the declared-storage fixture.");
        return true;
    }

    private bool VerifyStorageFixture()
    {
        var shop = EmporiumEntry.Instance;
        var item = shop.invElement.childItems.Concat(shop.backInvinvElement.childItems)
            .FirstOrDefault(candidate => GeneralHelper.GetCustomName(candidate) == StorageProbeName);
        if (item?.contentWindow == null) return false;
        var window = item.contentWindow;
        if (window.childElement is not MachineryStack stack || stack.elements.Count != 2 ||
            stack.elements[0] is not GameSlotInventory first || first.childItem != null ||
            stack.elements[1] is not GameSlotInventory second || second.childItem?.identifier != "small_bottled_water" ||
            !window.children.Contains(second) || !second.parents.Contains(window) ||
            !ReferenceEquals(second.childItem.parentInventory, second))
            throw new InvalidOperationException("The declared native slots lost their order or graph ownership.");
        if (!item.MayActivateSlotItem() || !item.CanActivateSlotItem())
            throw new InvalidOperationException("Original context activation was unavailable.");
        item.ActivateSlotItem();
        if (localActivations != 1 || CustomUIManager.Instance.GetWindow(ModId + ":local-view")?.IsAlive != true)
            throw new InvalidOperationException("Original context activation did not open one local window.");
        CustomUIManager.Instance.CloseWindow(ModId + ":local-view");
        Debug.Log("[coop-extension-probe] PASS original context activation opened one guest-only window");
        return true;
    }

    private bool OpenStorageView(GameItem item)
    {
        if (GeneralHelper.GetCustomName(item) != StorageProbeName) return false;
        localActivations++;
        CustomUIManager.Instance.CreateWindow(ModId + ":local-view", StorageProbeName).Show();
        return true;
    }

    private void ConfirmIncrement(CoopActionResult result)
    {
        if(result!=CoopActionResult.Applied || displayed!=1 || session.CurrentState?.Payload!="1")
        {
            Debug.LogError("[coop-extension-probe] FAIL increment acknowledgement "+result);
            phase=3;
            return;
        }
        phase=2;
        if(session.Request("increment","1",firstRevision,ConfirmStale)!=CoopActionSubmission.Queued)
        {
            Debug.LogError("[coop-extension-probe] FAIL stale request not queued");
            phase=3;
        }
    }

    private void ConfirmStale(CoopActionResult result)
    {
        if(result==CoopActionResult.Stale && displayed==1)
            Debug.Log("[coop-extension-probe] PASS native partner action, projected receipt and stale rejection");
        else Debug.LogError("[coop-extension-probe] FAIL stale request "+result);
        phase=3;
    }
}

internal sealed class DeclaredStorage : MachineryPanelRow, ICoopNativeStorageLayout
{
    public IReadOnlyList<GameInventory> CoopStorage { get; }

    internal DeclaredStorage()
    {
        var filled = new GameSlotInventory(2, 3);
        var empty = new GameSlotInventory(2, 3);
        // The visible order differs from the base row's insertion order.
        Add(new MachineryPanel("Filled", filled));
        Add(new MachineryPanel("Empty", empty));
        SetDisplayOrder(1, 0);
        CoopStorage = new GameInventory[] { empty, filled };
    }
}

public sealed class ProbePump : MonoBehaviour
{
    internal Action TickAction;
    private void Update() => TickAction?.Invoke();
}
